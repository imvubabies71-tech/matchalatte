local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")

local LocalPlayer = Players.LocalPlayer

local CONFIG = {
    MaxDistance = 1200,

    PlayerCacheInterval = 0.50,
    ItemCacheInterval = 0.20,
    AstralScanInterval = 0.20,

    CleanupInterval = 10,

    MaxDrawings = 500,

    CharacterOffset = 2.5,
    UsernameOffset = 17,
    ItemStartOffset = 32,
    ItemGap = 14,

    MapItemOffset = 1,
    AstralOffset = 3,
}

local COLORS = {
    Default = Color3.fromRGB(255, 255, 255),
    Username = Color3.fromRGB(180, 180, 180),
    Item = Color3.fromRGB(255, 255, 255),
    MapItem = Color3.fromRGB(255, 255, 255),
    Astral = Color3.fromRGB(180, 140, 255),
}

local SPECIE_COLORS = {
    Vampire = Color3.fromRGB(196, 30, 58),
    Werewolf = Color3.fromRGB(249, 228, 103),
    Witch = Color3.fromRGB(195, 145, 195),
    Original = Color3.fromRGB(154, 42, 42),
    Hybrid = Color3.fromRGB(245, 185, 102),
    Heretic = Color3.fromRGB(188, 101, 169),
    Mortal = Color3.fromRGB(193, 225, 193),
    Phoenix = Color3.fromRGB(223, 129, 96),
    Hunter = Color3.fromRGB(120, 199, 114),
    Tribrid = Color3.fromRGB(182, 208, 226),
    Immortal = Color3.fromRGB(126, 53, 248),
    Werewitch = Color3.fromRGB(201, 69, 150),
}

local ITEM_NAMES = {
    IndestructibleWhiteOakStake = "Indestructible",
    QetsiyahCure = "QetCure",
    TheCure = "Cure",
}

local ESP = {
    Enabled = true,
    CharacterName = true,
    Username = true,
    Items = true,
    MapItems = true,
    Astral = true,
}

local CachedPlayers = {}
local PlayerList = table.create(64)
local PlayerCount = 0

local IWOSCache = table.create(64)
local IWOSCount = 0

local CureCache = table.create(4)
local CureCount = 0

local AstralCache = {}
local AstralDrawings = {}

local CuredState = {}
local SpeciesCache = {}

local AstralProjection =
    Workspace:FindFirstChild("AstralProjection")

local lastPlayerCache = 0
local lastAstralScan = 0
local lastCleanup = 0
local now = 0

local MaxDistanceSquared =
    CONFIG.MaxDistance * CONFIG.MaxDistance


local function GetSpecieColor(specie)
    if specie == nil then
        return COLORS.Default
    end

    specie = tostring(specie)

    local cached = SpeciesCache[specie]

    if cached then
        return cached
    end

    local color = SPECIE_COLORS[specie]

    if not color then
        color = COLORS.Default
    end

    SpeciesCache[specie] = color

    return color
end


local function GetBasePart(object)
    if not object then
        return nil
    end

    if object:IsA("BasePart") then
        return object
    end

    if object:IsA("Model") then
        local primary = object.PrimaryPart

        if primary and primary:IsA("BasePart") then
            return primary
        end

        local root = object:FindFirstChild("HumanoidRootPart")

        if root and root:IsA("BasePart") then
            return root
        end

        local head = object:FindFirstChild("Head")

        if head and head:IsA("BasePart") then
            return head
        end

        return object:FindFirstChildWhichIsA(
            "BasePart",
            true
        )
    end

    return nil
end


local function IsInRange(position, origin)
    if not position then
        return false
    end

    local x = position.X - origin.X
    local y = position.Y - origin.Y
    local z = position.Z - origin.Z

    return
        x * x +
        y * y +
        z * z <= MaxDistanceSquared
end


local DrawPool = table.create(CONFIG.MaxDrawings)

for i = 1, CONFIG.MaxDrawings do
    local drawing = Drawing.new("Text")

    drawing.Font = Drawing.Fonts.SystemBold
    drawing.Center = true
    drawing.Outline = true
    drawing.Visible = false

    DrawPool[i] = drawing
end


local function HideAllDrawings()
    for i = 1, CONFIG.MaxDrawings do
        DrawPool[i].Visible = false
    end
end


local function SetDrawing(index, text, position, color)
    local drawing = DrawPool[index]

    if not drawing then
        return
    end

    drawing.Text = text
    drawing.Position = position
    drawing.Color = color
    drawing.Visible = true
end


local function UpdatePlayerItems(data, player, currentTime)
    if currentTime - data.LastItemUpdate <
        CONFIG.ItemCacheInterval
    then
        return
    end

    data.LastItemUpdate = currentTime

    local backpack = player:FindFirstChild("Backpack")

    if not backpack then
        data.ItemCount = 0
        return
    end

    local children = backpack:GetChildren()
    local count = 0

    for i = 1, #children do
        local itemName =
            ITEM_NAMES[children[i].Name]

        if itemName then
            count = count + 1
            data.Items[count] = itemName
        end
    end

    for i = count + 1, data.ItemCount do
        data.Items[i] = nil
    end

    data.ItemCount = count
end


local function UpdateIWOSCache()
    local folder =
        Workspace:FindFirstChild(
            "IndestructibleWhiteOakStake"
        )

    if not folder then
        for i = 1, IWOSCount do
            IWOSCache[i] = nil
        end

        IWOSCount = 0
        return
    end

    local children = folder:GetChildren()
    local count = 0

    for i = 1, #children do
        local part = GetBasePart(children[i])

        if part then
            count = count + 1
            IWOSCache[count] = part
        end
    end

    for i = count + 1, IWOSCount do
        IWOSCache[i] = nil
    end

    IWOSCount = count
end


local function UpdateCureCache()
    local interactables =
        Workspace:FindFirstChild("Interactables")

    if not interactables then
        CureCache[1] = nil
        CureCount = 0
        return
    end

    local silasTomb =
        interactables:FindFirstChild("SilasTomb")

    if not silasTomb then
        CureCache[1] = nil
        CureCount = 0
        return
    end

    local cureBox =
        silasTomb:FindFirstChild("CureBox")

    if not cureBox then
        CureCache[1] = nil
        CureCount = 0
        return
    end

    local cure =
        cureBox:FindFirstChild("TheCure")

    local part = GetBasePart(cure)

    if part then
        CureCache[1] = part
        CureCount = 1
    else
        CureCache[1] = nil
        CureCount = 0
    end
end


local function UpdatePlayerCache(currentTime)
    local players = Players:GetPlayers()
    local count = 0

    for i = 1, #players do
        local player = players[i]

        if player ~= LocalPlayer then
            local character = player.Character

            if character then
                local head =
                    character:FindFirstChild("Head")

                if head and head:IsA("BasePart") then
                    count = count + 1

                    local data =
                        CachedPlayers[player]

                    if not data then
                        data = {
                            Character = character,
                            Head = head,
                            Root = nil,

                            CharacterName = player.Name,
                            Username = player.Name,

                            SpecieColor =
                                COLORS.Default,

                            Items = table.create(4),
                            ItemCount = 0,

                            LastItemUpdate = 0,
                        }

                        CachedPlayers[player] = data
                    end

                    data.Character = character
                    data.Head = head

                    local root =
                        character:FindFirstChild(
                            "HumanoidRootPart"
                        )

                    if root and root:IsA("BasePart") then
                        data.Root = root
                    else
                        data.Root = nil
                    end

                    local characterName =
                        player:GetAttribute(
                            "CharacterName"
                        )

                    if characterName ~= nil
                        and tostring(characterName) ~= ""
                    then
                        data.CharacterName =
                            tostring(characterName)
                    else
                        data.CharacterName =
                            player.Name
                    end

                    local specie =
                        character:GetAttribute(
                            "SpecieType"
                        )

                    data.SpecieColor =
                        GetSpecieColor(specie)

                    data.Username = player.Name

                    UpdatePlayerItems(
                        data,
                        player,
                        currentTime
                    )

                    PlayerList[count] = player
                end
            end
        end
    end

    for i = count + 1, PlayerCount do
        PlayerList[i] = nil
    end

    PlayerCount = count

    UpdateIWOSCache()
    UpdateCureCache()
end


local function FindAstralPart(model)
    if not model then
        return nil
    end

    if not model.Parent then
        return nil
    end

    local root =
        model:FindFirstChild("HumanoidRootPart")

    if root and root:IsA("BasePart") then
        return root
    end

    local head =
        model:FindFirstChild("Head")

    if head and head:IsA("BasePart") then
        return head
    end

    local primary = model.PrimaryPart

    if primary and primary:IsA("BasePart") then
        return primary
    end

    return model:FindFirstChildWhichIsA(
        "BasePart",
        true
    )
end


local function CreateAstralDrawing(model)
    local drawing =
        AstralDrawings[model]

    if drawing then
        return drawing
    end

    drawing = Drawing.new("Text")

    drawing.Font =
        Drawing.Fonts.SystemBold

    drawing.Center = true
    drawing.Outline = true
    drawing.Text = "[Astral Body]"
    drawing.Color = COLORS.Astral
    drawing.Visible = false

    AstralDrawings[model] = drawing

    return drawing
end


local function RemoveAstralDrawing(model)
    local drawing =
        AstralDrawings[model]

    if drawing then
        drawing.Visible = false
        drawing:Remove()
        AstralDrawings[model] = nil
    end
end


local function UpdateAstralCache()
    if not AstralProjection
        or not AstralProjection.Parent
    then
        AstralProjection =
            Workspace:FindFirstChild(
                "AstralProjection"
            )

        if not AstralProjection then
            return
        end
    end

    local children =
        AstralProjection:GetChildren()

    local active = {}

    for i = 1, #children do
        local model = children[i]

        if model:IsA("Model") then
            local part =
                FindAstralPart(model)

            if part then
                AstralCache[model] = part
                active[model] = true

                CreateAstralDrawing(model)
            end
        end
    end

    for model in next, AstralCache do
        if not active[model]
            or not model.Parent
        then
            AstralCache[model] = nil
            RemoveAstralDrawing(model)
        else
            local part =
                AstralCache[model]

            if not part
                or not part.Parent
            then
                AstralCache[model] =
                    FindAstralPart(model)
            end
        end
    end
end


local function UpdateCuredState()
    local players = Players:GetPlayers()

    for i = 1, #players do
        local player = players[i]

        local cured =
            player:GetAttribute("Cured") == true

        local id = player.UserId
        local previous = CuredState[id]

        if previous == nil then
            CuredState[id] = cured

        elseif previous ~= cured then
            CuredState[id] = cured

            if cured then
                local name =
                    player:GetAttribute(
                        "CharacterName"
                    )

                if name == nil
                    or tostring(name) == ""
                then
                    name = player.Name
                end

                name = tostring(name):gsub(
                    "^%s*(.-)%s*$",
                    "%1"
                )

                notify(
                    name .. " has been cured!",
                    "Cured",
                    4
                )
            end
        end
    end
end


local function Cleanup()
    for player in next, CachedPlayers do
        if not player.Parent then
            CachedPlayers[player] = nil
        end
    end

    local players =
        Players:GetPlayers()

    local activePlayers = {}

    for i = 1, #players do
        activePlayers[
            players[i].UserId
        ] = true
    end

    for id in next, CuredState do
        if not activePlayers[id] then
            CuredState[id] = nil
        end
    end

    table.clear(SpeciesCache)
end


UpdatePlayerCache(0)
UpdateAstralCache()
UpdateCuredState()

notify(
    "Dav's Hub loaded successfully!",
    "Loaded",
    4
)


RunService.RenderStepped:Connect(function(deltaTime)
    now = now + deltaTime

    if now - lastPlayerCache >=
        CONFIG.PlayerCacheInterval
    then
        lastPlayerCache = now
        UpdatePlayerCache(now)
        UpdateCuredState()
    end

    if now - lastAstralScan >=
        CONFIG.AstralScanInterval
    then
        lastAstralScan = now

        if ESP.Astral then
            UpdateAstralCache()
        end
    end

    if now - lastCleanup >=
        CONFIG.CleanupInterval
    then
        lastCleanup = now
        Cleanup()
    end

    if not ESP.Enabled then
        HideAllDrawings()
        return
    end

    local localCharacter =
        LocalPlayer.Character

    if not localCharacter then
        HideAllDrawings()
        return
    end

    local localRoot =
        localCharacter:FindFirstChild(
            "HumanoidRootPart"
        )

    if not localRoot
        or not localRoot:IsA("BasePart")
    then
        HideAllDrawings()
        return
    end

    local origin =
        localRoot.Position

    local used = 0


    -- PLAYER ESP

    for i = 1, PlayerCount do
        if used >= CONFIG.MaxDrawings - 10 then
            break
        end

        local player =
            PlayerList[i]

        local data =
            CachedPlayers[player]

        if data then
            local head = data.Head

            if head
                and head.Parent
                and head:IsA("BasePart")
            then
                local position =
                    head.Position

                if IsInRange(
                    position,
                    origin
                ) then
                    local screen, visible =
                        WorldToScreen(
                            position +
                            Vector3.new(
                                0,
                                CONFIG.CharacterOffset,
                                0
                            )
                        )

                    if visible then
                        local x = screen.X
                        local y = screen.Y

                        if ESP.CharacterName then
                            used = used + 1

                            SetDrawing(
                                used,
                                data.CharacterName,
                                Vector2.new(
                                    x,
                                    y
                                ),
                                data.SpecieColor
                            )
                        end

                        if ESP.Username
                            and used <
                                CONFIG.MaxDrawings
                        then
                            used = used + 1

                            SetDrawing(
                                used,
                                "@" ..
                                    data.Username,
                                Vector2.new(
                                    x,
                                    y +
                                    CONFIG.UsernameOffset
                                ),
                                COLORS.Username
                            )
                        end

                        if ESP.Items
                            and data.ItemCount > 0
                        then
                            local itemY =
                                y +
                                CONFIG.ItemStartOffset

                            for j = 1,
                                data.ItemCount
                            do
                                if used >=
                                    CONFIG.MaxDrawings
                                then
                                    break
                                end

                                used = used + 1

                                SetDrawing(
                                    used,
                                    "[" ..
                                        data.Items[j] ..
                                        "]",
                                    Vector2.new(
                                        x,
                                        itemY
                                    ),
                                    COLORS.Item
                                )

                                itemY =
                                    itemY +
                                    CONFIG.ItemGap
                            end
                        end
                    end
                end
            end
        end
    end


    -- MAP ITEMS

    if ESP.MapItems then

        for i = 1, IWOSCount do
            if used >= CONFIG.MaxDrawings then
                break
            end

            local part =
                IWOSCache[i]

            if part
                and part.Parent
                and part:IsA("BasePart")
            then
                local position =
                    part.Position

                if IsInRange(
                    position,
                    origin
                ) then
                    local screen, visible =
                        WorldToScreen(
                            position +
                            Vector3.new(
                                0,
                                CONFIG.MapItemOffset,
                                0
                            )
                        )

                    if visible then
                        used = used + 1

                        SetDrawing(
                            used,
                            "[Indestructible]",
                            screen,
                            COLORS.MapItem
                        )
                    end
                end
            end
        end


        for i = 1, CureCount do
            if used >= CONFIG.MaxDrawings then
                break
            end

            local part =
                CureCache[i]

            if part
                and part.Parent
                and part:IsA("BasePart")
            then
                local position =
                    part.Position

                if IsInRange(
                    position,
                    origin
                ) then
                    local screen, visible =
                        WorldToScreen(
                            position +
                            Vector3.new(
                                0,
                                CONFIG.MapItemOffset,
                                0
                            )
                        )

                    if visible then
                        used = used + 1

                        SetDrawing(
                            used,
                            "[Cure]",
                            screen,
                            COLORS.MapItem
                        )
                    end
                end
            end
        end
    end


    -- ASTRAL PROJECTION

    if ESP.Astral then
        for model, part in next, AstralCache do
            local drawing =
                AstralDrawings[model]

            if drawing then
                if not model.Parent then
                    drawing.Visible = false
                elseif not part
                    or not part.Parent
                    or not part:IsA("BasePart")
                then
                    drawing.Visible = false
                else
                    local position =
                        part.Position

                    if IsInRange(
                        position,
                        origin
                    ) then
                        local screen, visible =
                            WorldToScreen(
                                position +
                                Vector3.new(
                                    0,
                                    CONFIG.AstralOffset,
                                    0
                                )
                            )

                        if visible then
                            drawing.Position =
                                screen

                            drawing.Visible = true
                        else
                            drawing.Visible = false
                        end
                    else
                        drawing.Visible = false
                    end
                end
            end
        end
    else
        for _, drawing in next,
            AstralDrawings
        do
            drawing.Visible = false
        end
    end


    -- HIDE UNUSED PLAYER/MAP DRAWINGS

    for i = used + 1,
        CONFIG.MaxDrawings
    do
        if DrawPool[i].Visible then
            DrawPool[i].Visible = false
        end
    end
end)
