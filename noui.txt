local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local lp = Players.LocalPlayer

-- ---- SETTINGS (ca la motorul mic) --------------------------
local POOL         = 500      -- max targets
local COLLECT_RATE = 0.5      -- seconds between collector ticks
local SHOW_SELF    = false

-- ---- CONFIG (din original) --------------------------------
local CONFIG = {
    MaxDistance = 1200,
    MaxDrawings = 500,
    CharacterOffset = 2.5,
    UsernameOffset = 17,
    ItemStartOffset = 32,
    ItemGap = 14,
    MapItemOffset = 1,
    AstralOffset = 3,
}

-- ---- COLORS (din original) --------------------------------
local COLORS = {
    Default = Color3.fromRGB(255, 255, 255),
    Username = Color3.fromRGB(200, 200, 200),
    Item = Color3.fromRGB(255, 255, 255),
    MapItem = Color3.fromRGB(255, 255, 255),
    Astral = Color3.fromRGB(180, 140, 255),
}

local SPECIE_COLORS = {
    Vampire = Color3.fromRGB(196, 30, 58),
    Werewolf = Color3.fromRGB(249, 228, 103),
    Witch = Color3.fromRGB(195, 145, 195),
    Original = Color3.fromRGB(154, 42, 42),
    Hybrid = Color3.fromRGB(245, 185, 102),
    Heretic = Color3.fromRGB(188, 101, 169),
    Mortal = Color3.fromRGB(193, 225, 193),
    Phoenix = Color3.fromRGB(223, 129, 96),
    Hunter = Color3.fromRGB(120, 199, 114),
    Tribrid = Color3.fromRGB(182, 208, 226),
    Immortal = Color3.fromRGB(126, 53, 248),
    Werewitch = Color3.fromRGB(201, 69, 150),
}

local ITEM_NAMES = {
    IndestructibleWhiteOakStake = "Indestructible",
    QetsiyahCure = "QetCure",
    TheCure = "Cure",
}

-- ---- ESP TOGGLES -------------------------------------------
local ESP = {
    Enabled = true,
    CharacterName = true,
    Username = true,
    Items = true,
    MapItems = true,
    Astral = true,
    BodyJumped = true,
}

-- ---- STATE (ca la motorul mic) -----------------------------
local running = true
local targets = {}
local SpeciesCache = {}

-- ---- CACHES (din original) --------------------------------
local IWOSCache = {}
local IWOSCount = 0
local CureCache = {}
local CureCount = 0
local AstralCache = {}
local AstralDrawings = {}
local CuredState = {}
local BodyJumpedState = {}
local AstralProjection = Workspace:FindFirstChild("AstralProjection")

-- ---- POOL SETUP (ca la motorul mic) -----------------------
local pool = {}
for i = 1, POOL do
    local label = Drawing.new("Text")
    label.Font = Drawing.Fonts.SystemBold
    label.Size = 14
    label.Center = true
    label.Outline = true
    label.Color = COLORS.Default
    label.Visible = false
    pool[i] = label
end

-- ---- HELPERS (din original) -------------------------------
local function GetSpecieColor(specie)
    if not specie then return COLORS.Default end
    specie = tostring(specie)
    local cached = SpeciesCache[specie]
    if cached then return cached end
    local color = SPECIE_COLORS[specie] or COLORS.Default
    SpeciesCache[specie] = color
    return color
end

local function GetBasePart(object)
    if not object then return nil end
    if object:IsA("BasePart") then return object end
    if object:IsA("Model") then
        local primary = object.PrimaryPart
        if primary and primary:IsA("BasePart") then return primary end
        local root = object:FindFirstChild("HumanoidRootPart")
        if root and root:IsA("BasePart") then return root end
        local head = object:FindFirstChild("Head")
        if head and head:IsA("BasePart") then return head end
        return object:FindFirstChildWhichIsA("BasePart", true)
    end
    return nil
end

local maxDistSq = CONFIG.MaxDistance * CONFIG.MaxDistance
local function IsInRange(position, origin)
    if not position then return false end
    local x = position.X - origin.X
    local y = position.Y - origin.Y
    local z = position.Z - origin.Z
    return x*x + y*y + z*z <= maxDistSq
end

-- ---- ITEMS FUNCTIONS (din original) ------------------------
local function UpdatePlayerItems(data, player)
    local backpack = player:FindFirstChild("Backpack")
    if not backpack then data.ItemCount = 0 return end
    local children = backpack:GetChildren()
    local count = 0
    for i = 1, #children do
        local itemName = ITEM_NAMES[children[i].Name]
        if itemName then
            count = count + 1
            data.Items[count] = itemName
        end
    end
    for i = count + 1, data.ItemCount do
        data.Items[i] = nil
    end
    data.ItemCount = count
end

local function UpdateIWOSCache()
    local folder = Workspace:FindFirstChild("IndestructibleWhiteOakStake")
    if not folder then
        IWOSCount = 0
        return
    end
    local children = folder:GetChildren()
    local count = 0
    for i = 1, #children do
        local part = GetBasePart(children[i])
        if part then
            count = count + 1
            IWOSCache[count] = part
        end
    end
    IWOSCount = count
end

local function UpdateCureCache()
    local interactables = Workspace:FindFirstChild("Interactables")
    if not interactables then CureCount = 0 return end
    local silasTomb = interactables:FindFirstChild("SilasTomb")
    if not silasTomb then CureCount = 0 return end
    local cureBox = silasTomb:FindFirstChild("CureBox")
    if not cureBox then CureCount = 0 return end
    local cure = cureBox:FindFirstChild("TheCure")
    local part = GetBasePart(cure)
    if part then
        CureCache[1] = part
        CureCount = 1
    else
        CureCount = 0
    end
end

-- ---- ASTRAL FUNCTIONS (din original) -----------------------
local function FindAstralPart(model)
    if not model or not model.Parent then return nil end
    local root = model:FindFirstChild("HumanoidRootPart")
    if root and root:IsA("BasePart") then return root end
    local head = model:FindFirstChild("Head")
    if head and head:IsA("BasePart") then return head end
    local primary = model.PrimaryPart
    if primary and primary:IsA("BasePart") then return primary end
    return model:FindFirstChildWhichIsA("BasePart", true)
end

local function CreateAstralDrawing(model)
    local drawing = AstralDrawings[model]
    if drawing then return drawing end
    drawing = Drawing.new("Text")
    drawing.Font = Drawing.Fonts.SystemBold
    drawing.Center = true
    drawing.Outline = true
    drawing.Text = "[Astral Body]"
    drawing.Color = COLORS.Astral
    drawing.Size = 13
    drawing.Visible = false
    AstralDrawings[model] = drawing
    return drawing
end

local function UpdateAstralCache()
    if not AstralProjection or not AstralProjection.Parent then
        AstralProjection = Workspace:FindFirstChild("AstralProjection")
        if not AstralProjection then return end
    end
    local children = AstralProjection:GetChildren()
    local active = {}
    for i = 1, #children do
        local model = children[i]
        if model:IsA("Model") then
            local part = FindAstralPart(model)
            if part then
                AstralCache[model] = part
                active[model] = true
                CreateAstralDrawing(model)
            end
        end
    end
    for model in next, AstralCache do
        if not active[model] or not model.Parent then
            AstralCache[model] = nil
            local drawing = AstralDrawings[model]
            if drawing then
                drawing.Visible = false
                drawing:Remove()
                AstralDrawings[model] = nil
            end
        else
            local part = AstralCache[model]
            if not part or not part.Parent then
                AstralCache[model] = FindAstralPart(model)
            end
        end
    end
end

-- ---- CURED STATE (din original) ----------------------------
local function UpdateCuredState()
    local players = Players:GetPlayers()
    for i = 1, #players do
        local player = players[i]
        local cured = player:GetAttribute("Cured") == true
        local id = player.UserId
        local previous = CuredState[id]
        if previous == nil then
            CuredState[id] = cured
        elseif previous ~= cured then
            CuredState[id] = cured
            if cured then
                local name = player:GetAttribute("CharacterName")
                if name == nil or tostring(name) == "" then
                    name = player.Name
                end
                name = tostring(name):gsub("^%s*(.-)%s*$", "%1")
                notify(name .. " has been cured!", "Cured", 4)
            end
        end
    end
end

-- ---- BODYJUMPED STATE ---------------------------------------
local function UpdateBodyJumpedState()
    local players = Players:GetPlayers()
    for i = 1, #players do
        local player = players[i]
        local bodyJumped = player:GetAttribute("BodyJumped") == true
        local id = player.UserId
        local previous = BodyJumpedState[id]
        
        if previous == nil then
            BodyJumpedState[id] = bodyJumped
        elseif previous ~= bodyJumped then
            BodyJumpedState[id] = bodyJumped
            if bodyJumped then
                local name = player:GetAttribute("CharacterName")
                if name == nil or tostring(name) == "" then
                    name = player.Name
                end
                name = tostring(name):gsub("^%s*(.-)%s*$", "%1")
                notify(name .. " has been bodyjumped by Esther Mikaleson!", "BodyJumped", 4)
            end
        end
    end
end

-- ---- CLEANUP (din original) --------------------------------
local function Cleanup()
    local players = Players:GetPlayers()
    local activePlayers = {}
    for i = 1, #players do
        activePlayers[players[i].UserId] = true
    end
    for id in next, CuredState do
        if not activePlayers[id] then
            CuredState[id] = nil
        end
    end
    for id in next, BodyJumpedState do
        if not activePlayers[id] then
            BodyJumpedState[id] = nil
        end
    end
    table.clear(SpeciesCache)
end

-- ---- COLLECTOR (exact ca la motorul mic) --------------------
task.spawn(function()
    while running do
        local out = {}
        local players = Players:GetPlayers()
        local origin = nil
        local localChar = lp.Character
        if localChar then
            local localRoot = localChar:FindFirstChild("HumanoidRootPart")
            if localRoot and localRoot:IsA("BasePart") then
                origin = localRoot.Position
            end
        end
        
        -- Update all caches
        UpdateIWOSCache()
        UpdateCureCache()
        UpdateAstralCache()
        UpdateCuredState()
        UpdateBodyJumpedState()
        Cleanup()

        for i = 1, #players do
            local plr = players[i]
            if SHOW_SELF or plr ~= lp then
                local char = plr.Character
                local head = char and char:FindFirstChild("Head")
                if head and head:IsA("BasePart") then
                    if not origin or IsInRange(head.Position, origin) then
                        -- Get attributes
                        local charName = plr:GetAttribute("CharacterName")
                        local specie = char:GetAttribute("SpecieType")
                        local isBodyJumped = plr:GetAttribute("BodyJumped") == true
                        
                        -- Culoarea rămâne cea a speciei
                        local displayColor = GetSpecieColor(specie)
                        
                        -- Build label text - if bodyjumped, show "Esther Mikaelson"
                        local labelText
                        if isBodyJumped and ESP.BodyJumped then
                            labelText = "Esther Mikaelson"
                        else
                            labelText = (charName and charName ~= "") and charName or plr.Name
                        end
                        
                        -- Get items
                        local itemData = { Items = {}, ItemCount = 0 }
                        UpdatePlayerItems(itemData, plr)
                        
                        out[#out + 1] = {
                            head = head,
                            text = labelText,
                            color = displayColor,
                            player = plr,
                            items = itemData.Items,
                            itemCount = itemData.ItemCount,
                            isBodyJumped = isBodyJumped,
                        }
                    end
                end
            end
        end
        
        targets = out
        task.wait(COLLECT_RATE)
    end
end)

-- ---- RENDERER (exact ca la motorul mic) ---------------------
local conn
conn = RunService.RenderStepped:Connect(function()
    if not ESP.Enabled then
        for i = 1, POOL do pool[i].Visible = false end
        for _, drawing in next, AstralDrawings do drawing.Visible = false end
        return
    end

    local localChar = lp.Character
    if not localChar then
        for i = 1, POOL do pool[i].Visible = false end
        return
    end

    local localRoot = localChar:FindFirstChild("HumanoidRootPart")
    if not localRoot or not localRoot:IsA("BasePart") then
        for i = 1, POOL do pool[i].Visible = false end
        return
    end

    local origin = localRoot.Position
    local list = targets
    local n = 0
    local maxDrawings = CONFIG.MaxDrawings

    for i = 1, #list do
        if n >= POOL then break end
        local t = list[i]
        
        if t.head and t.head.Parent then
            local headPos = t.head.Position
            
            if IsInRange(headPos, origin) then
                local pos3D = headPos + Vector3.new(0, CONFIG.CharacterOffset, 0)
                local pos2D, visible = WorldToScreen(pos3D)
                
                if visible then
                    local x = pos2D.X
                    local y = pos2D.Y
                    
                    -- Character Name (size 14)
                    if ESP.CharacterName then
                        n = n + 1
                        local lbl = pool[n]
                        lbl.Text = t.text
                        lbl.Color = t.color
                        lbl.Position = pos2D
                        lbl.Size = 14
                        lbl.Visible = true
                    end
                    
                    -- Username (size 13)
                    if ESP.Username and n < POOL then
                        n = n + 1
                        local lbl = pool[n]
                        lbl.Text = "@" .. t.player.Name
                        lbl.Color = COLORS.Username
                        lbl.Position = Vector2.new(x, y + CONFIG.UsernameOffset)
                        lbl.Size = 13
                        lbl.Visible = true
                    end
                    
                    -- Items (size 12)
                    if ESP.Items and t.itemCount > 0 then
                        local itemY = y + CONFIG.ItemStartOffset
                        for j = 1, t.itemCount do
                            if n >= POOL then break end
                            n = n + 1
                            local lbl = pool[n]
                            lbl.Text = "[" .. t.items[j] .. "]"
                            lbl.Color = COLORS.Item
                            lbl.Position = Vector2.new(x, itemY)
                            lbl.Size = 12
                            lbl.Visible = true
                            itemY = itemY + CONFIG.ItemGap
                        end
                    end
                end
            end
        end
    end
    
    -- Map Items (size 12)
    if ESP.MapItems then
        for i = 1, IWOSCount do
            if n >= POOL then break end
            local part = IWOSCache[i]
            if part and part.Parent and part:IsA("BasePart") then
                if IsInRange(part.Position, origin) then
                    local pos2D, visible = WorldToScreen(part.Position + Vector3.new(0, CONFIG.MapItemOffset, 0))
                    if visible then
                        n = n + 1
                        local lbl = pool[n]
                        lbl.Text = "[Indestructible]"
                        lbl.Color = COLORS.MapItem
                        lbl.Position = pos2D
                        lbl.Size = 12
                        lbl.Visible = true
                    end
                end
            end
        end
        
        for i = 1, CureCount do
            if n >= POOL then break end
            local part = CureCache[i]
            if part and part.Parent and part:IsA("BasePart") then
                if IsInRange(part.Position, origin) then
                    local pos2D, visible = WorldToScreen(part.Position + Vector3.new(0, CONFIG.MapItemOffset, 0))
                    if visible then
                        n = n + 1
                        local lbl = pool[n]
                        lbl.Text = "[Cure]"
                        lbl.Color = COLORS.MapItem
                        lbl.Position = pos2D
                        lbl.Size = 12
                        lbl.Visible = true
                    end
                end
            end
        end
    end
    
    -- Astral (size 13)
    if ESP.Astral then
        for model, part in next, AstralCache do
            local drawing = AstralDrawings[model]
            if drawing then
                if not model.Parent or not part or not part.Parent or not part:IsA("BasePart") then
                    drawing.Visible = false
                else
                    if IsInRange(part.Position, origin) then
                        local pos2D, visible = WorldToScreen(part.Position + Vector3.new(0, CONFIG.AstralOffset, 0))
                        if visible then
                            drawing.Position = pos2D
                            drawing.Size = 13
                            drawing.Visible = true
                        else
                            drawing.Visible = false
                        end
                    else
                        drawing.Visible = false
                    end
                end
            end
        end
    else
        for _, drawing in next, AstralDrawings do
            drawing.Visible = false
        end
    end
    
    -- Hide unused slots
    for i = n + 1, POOL do
        pool[i].Visible = false
    end
end)

-- ---- STOP FUNCTION (ca la motorul mic) ----------------------
_G.ESPStop = function()
    running = false
    if conn then conn:Disconnect() end
    for i = 1, POOL do pool[i]:Remove() end
    for _, drawing in next, AstralDrawings do drawing:Remove() end
    _G.ESPStop = nil
    print("ESP stopped.")
end
